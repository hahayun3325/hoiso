    66	def run_hunyuan_w_guid(
    67	    cropped_obj_img_path,
    68	    fovx,
    69	    hamer_for_guid_path,
    70	    aligned_mano_mesh_path,
    71	    cropped_obj_mask_path,
    72	    cropped_hand_mask_path,
    73	    moge_mesh_path,
    74	    T_h2m_path,
    75	    hunyuan_hoi_mesh_path,
    76	    save_path_obj,
    77	    save_path_hand,
    78	    config,
    79	    device="cuda",
    80	):
    81	    original_img_hand_mask = cv2.imread(cropped_hand_mask_path, cv2.IMREAD_GRAYSCALE)
    82	    img_size = original_img_hand_mask.shape
    83	    H, W = img_size[:2]
    84	
    85	    render_scale = float(os.environ.get("FOHO_RENDER_SCALE", "1.0"))
    86	    if render_scale != 1.0:
    87	        print(
    88	            f"[FOHO_LOW_MEM][WARN] FOHO_RENDER_SCALE={render_scale} is ignored for now "
    89	            "because renderer outputs must match MoGe/mask resolution."
    90	        )
    91	
    92	    render_faces_per_pixel = int(os.environ.get("FOHO_RENDER_FACES_PER_PIXEL", "1"))
    93	    sil_faces_per_pixel = int(os.environ.get("FOHO_SIL_FACES_PER_PIXEL", "100"))
    94	    print(f"[FOHO_LOW_MEM] renderer image_size=({H}, {W})")
    95	    print(f"[FOHO_LOW_MEM] render_faces_per_pixel={render_faces_per_pixel}, sil_faces_per_pixel={sil_faces_per_pixel}")
    96	
    97	    rotation_y_180 = torch.tensor(
    98	        [[-1, 0, 0], [0, 1, 0], [0, 0, -1]], device=device, dtype=torch.float32
    99	    )
   100	    fov = fovx
   101	    R = rotation_y_180.unsqueeze(0)
   102	    T = torch.zeros(1, 3, device=device)
   103	    cameras = FoVPerspectiveCameras(device=device, R=R, T=T, znear=0.01, zfar=100.0, fov=fov)
   104	    blend_params = BlendParams(
   105	        sigma=torch.tensor(1e-8, dtype=torch.float32, device=device),
   106	        gamma=torch.tensor(1e-8, dtype=torch.float32, device=device),
   107	    )
   108	    raster_settings = RasterizationSettings(
   109	        image_size=(H, W),
   110	        blur_radius=np.log(1.0 / 1e-4 - 1.0) * blend_params.sigma,
   111	        faces_per_pixel=render_faces_per_pixel,
   112	        bin_size=-1,
   113	        max_faces_per_bin=None,
   114	    )
   115	    renderer = MeshRenderer(
   116	        rasterizer=MeshRasterizer(cameras=cameras, raster_settings=raster_settings),
   117	        shader=PhongNormalShader(cameras=cameras, blend_params=blend_params),
   118	    )
   119	    silhoutte_raster_settings = RasterizationSettings(
   120	        image_size=(H, W),
   121	        blur_radius=np.log(1.0 / 1e-4 - 1.0) * blend_params.sigma,
   122	        faces_per_pixel=sil_faces_per_pixel,
   123	        bin_size=None,
   124	        max_faces_per_bin=None,
   125	    )
   126	    silhouette_renderer = MeshRenderer(
   127	        rasterizer=MeshRasterizer(cameras=cameras, raster_settings=silhoutte_raster_settings),
   128	        shader=SoftSilhouetteShader(blend_params=blend_params),
   129	    )
   130	
   131	    model_path = "tencent/Hunyuan3D-2"
   132	    rembg = BackgroundRemover()
   133	    seed = 2 #2025
   134	
   135	    if "inpaint" in cropped_obj_img_path:
   136	        images = []
   137	        image = Image.open(cropped_obj_img_path).convert("RGB")
   138	        if image.mode == "RGB":
   139	            image = rembg(image)
   140	        images.append(image)
   141	    else:
   142	        image = Image.open(cropped_obj_img_path).convert("RGBA")
   143	        datas = image.getdata()
   144	        new_data = []
   145	        for item in datas:
   146	            if item[0] == 255 and item[1] == 255 and item[2] == 255:
   147	                new_data.append((255, 255, 255, 0))
   148	            else:
   149	                new_data.append(item)
   150	        image.putdata(new_data)
   151	        images = [image]
   152	
   153	    pipeline = Hunyuan3DDiTFlowMatchingPipeline_main.from_pretrained(model_path)
   154	    obj_mesh, hand_mesh = pipeline(
   155	        image=images,
   156	        mc_algo="mc",
   157	        generator=torch.manual_seed(seed),
   158	        config=config,
   159	        renderer=renderer,
   160	        sil_renderer=silhouette_renderer,
   161	        cropped_obj_img_path=cropped_obj_img_path,
   162	        hamer_for_guid_path=hamer_for_guid_path,
   163	        aligned_mano_mesh_path=aligned_mano_mesh_path,
   164	        obj_mask_path=cropped_obj_mask_path,
   165	        hand_mask_path=cropped_hand_mask_path,
   166	        moge_mesh_path=moge_mesh_path,
   167	        h2m_rt_path=T_h2m_path,
   168	        hunyuan_hoi_mesh_path=hunyuan_hoi_mesh_path,
   169	    )
   170	
   171	    try:
   172	        obj_mesh_verts, obj_mesh_faces = obj_mesh.verts_packed(), obj_mesh.faces_packed()
   173	        obj_mesh = trimesh.Trimesh(vertices=obj_mesh_verts.cpu().numpy(), faces=obj_mesh_faces.cpu().numpy())
   174	        obj_mesh = FloaterRemover()(obj_mesh)
   175	        obj_mesh = DegenerateFaceRemover()(obj_mesh)
   176	        obj_mesh = FaceReducer()(obj_mesh)
   177	        obj_mesh.export(save_path_obj)
   178	
   179	        IO().save_mesh(hand_mesh, save_path_hand)
   180	    except Exception:
   181	        print(f"Error in saving mesh for {cropped_obj_img_path}")
   182	        return None, None
   183	
   184	    if len(obj_mesh.vertices) == 0:
   185	        print(f"Empty mesh for {cropped_obj_img_path}")
   186	        return None, None
   187	
   188	    return obj_mesh, hand_mesh

===== guidance loop and save region =====
   201	def run(
   202	    project_root: str,
   203	    cropped_obj_img_dir: str,
   204	    mask_dir: str,
   205	    moge_out_dir: str,
   206	    hunyuan_hoi_mesh_dir: str,
   207	    hamer_out_dir: str,
   208	    h2m_rt_dir: str,
   209	    aligned_mano_dir: str,
   210	    guidance_out_dir: str,
   211	    task_list_file: Optional[str] = None,
   212	) -> None:
   213	    _setup_sys_path(project_root)
   214	
   215	    config = OptimizationConfig()
   216	
   217	    os.makedirs(guidance_out_dir, exist_ok=True)
   218	
   219	    assigned_imgs = _load_task_list(task_list_file, cropped_obj_img_dir)
   220	
   221	    for cropped_obj_img in tqdm(assigned_imgs):
   222	        try:
   223	            cropped_obj_img_path = os.path.join(cropped_obj_img_dir, cropped_obj_img)
   224	            index = cropped_obj_img.split("_")[0]
   225	            is_right = cropped_obj_img.split("_")[-1].split(".")[0]
   226	            cropped_hand_mask_path = os.path.join(mask_dir, f"{index}_cropped_hand_mask.png")
   227	            cropped_obj_mask_path = os.path.join(mask_dir, f"{index}_cropped_obj_mask.png")
   228	            moge_mesh_path = os.path.join(moge_out_dir, f"{index}_cropped_hoi/mesh.glb")
   229	            moge_fov_path = os.path.join(moge_out_dir, f"{index}_cropped_hoi/fov.json")
   230	            T_h2m_path = os.path.join(h2m_rt_dir, f"{index}_hoi_mesh.npy")
   231	            aligned_mano_mesh_path = os.path.join(aligned_mano_dir, f"{index}_hamer_aligned_mano.ply")
   232	            hunyuan_hoi_mesh_path = os.path.join(hunyuan_hoi_mesh_dir, f"{index}_hoi_mesh.ply")
   233	            hamer_for_guid_path = os.path.join(hamer_out_dir, f"{index}_kps_for_guidance.npy")
   234	            save_path_obj = os.path.join(guidance_out_dir, f"{index}_obj.ply")
   235	            save_path_hand = os.path.join(guidance_out_dir, f"{index}_hand.ply")
   236	
   237	            if os.path.exists(save_path_obj) and os.path.exists(save_path_hand):
   238	                print(f"{index} already exists, skipping")
   239	                continue
   240	
   241	            with open(moge_fov_path, "r", encoding="utf-8") as f:
   242	                moge_dict = json.load(f)
   243	            fovx = float(moge_dict["fov_x"])
   244	
   245	            cropped_hand_mask = cv2.imread(cropped_hand_mask_path, cv2.IMREAD_UNCHANGED)
   246	            cropped_obj_mask = cv2.imread(cropped_obj_mask_path, cv2.IMREAD_UNCHANGED)
   247	            if cropped_hand_mask.max() == 0 or cropped_obj_mask.max() == 0:
   248	                print(f"Skipping {index} due to empty mask")
   249	                continue
   250	
   251	            print(f"Processing {index}")
   252	            obj_mesh, hand_mesh = run_hunyuan_w_guid(
   253	                cropped_obj_img_path=cropped_obj_img_path,
   254	                fovx=fovx,
   255	                hamer_for_guid_path=hamer_for_guid_path,
   256	                aligned_mano_mesh_path=aligned_mano_mesh_path,
   257	                cropped_obj_mask_path=cropped_obj_mask_path,
   258	                cropped_hand_mask_path=cropped_hand_mask_path,
   259	                moge_mesh_path=moge_mesh_path,
   260	                T_h2m_path=T_h2m_path,
   261	                hunyuan_hoi_mesh_path=hunyuan_hoi_mesh_path,
   262	                save_path_obj=save_path_obj,
   263	                save_path_hand=save_path_hand,
   264	                config=config,
   265	            )
   266	            if obj_mesh is None or hand_mesh is None:
   267	                print(f"Error in reconstruction for {index}")
   268	                continue
   269	            print(f"Reconstructed object {index}")
   270	        except Exception as e:
   271	            print(f"Error in processing {cropped_obj_img} : {e}")
   272	            traceback.print_exc()
   273	            continue
   274	
   275	    print("Finished processing all images")
